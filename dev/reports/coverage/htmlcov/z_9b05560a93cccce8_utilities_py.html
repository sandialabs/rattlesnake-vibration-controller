79 """

187def _get_timezone_strings(utc_now: datetime) -> tuple[str, str, str, str]: 

188 """ 

189 Helper to get formatted UTC, EST, MST, and PST strings from a UTC datetime. 

190 

191 Args: 

192 utc_now: The UTC datetime object. 

193 

194 Returns: 

195 A tuple of four strings: (UTC string, EST string, MST string, PST string). 

196 """ 

197 # Define the time zones 

198 timezone_est: pytz.BaseTzInfo = pytz.timezone("America/New_York") 

199 timezone_mst: pytz.BaseTzInfo = pytz.timezone("America/Denver") 

200 timezone_pst: pytz.BaseTzInfo = pytz.timezone("America/Los_Angeles") 

201 

202 # Convert UTC time to EST, MST, and PST 

203 est_now: datetime = utc_now.astimezone(timezone_est) 

204 mst_now: datetime = utc_now.astimezone(timezone_mst) 

205 pst_now: datetime = utc_now.astimezone(timezone_pst) 

206 

207 # Format the output 

208 df: str = "%Y-%m-%d %H:%M:%S" # Date format 

209 utc_str: str = utc_now.strftime(f"{df} UTC") 

210 est_str: str = est_now.strftime(f"{df} EST") 

211 mst_str: str = mst_now.strftime(f"{df} MST") 

212 pst_str: str = pst_now.strftime(f"{df} PST") 

213 

214 return utc_str, est_str, mst_str, pst_str

217def get_timestamp() -> str: 

218 """ 

219 Get formatted timestamp with UTC, EST, MST, and PST times. 

220 

221 Returns: 

222 Formatted timestamp string 

223 """ 

224 # Get the current UTC time 

225 utc_now: datetime = datetime.now(pytz.utc) 

226 

227 # Get the formatted strings for each timezone 

228 utc_str, est_str, mst_str, pst_str = _get_timezone_strings(utc_now) 

229 

230 # Combine the formatted times 

231 timestamp: str = f"{utc_str} ({est_str} / {mst_str} / {pst_str})" 

232 

233 return timestamp

236def extend_timestamp(short: str) -> str: 

237 """ 

238 Given a timestamp string from CI/CD in the form of 

239 20250815_211112_UTC, extend the timestamp to include EST, MST, and PST times 

240 and return the extended string. 

241 

242 Args: 

243 short: the UTC bash string, for example: 20250815_211112_UTC 

244 

245 Returns: 

246 Extended timestamp string. 

247 """ 

248 # Call the multiline function to get the individual strings 

249 lines: list[str] = get_multiline_timestamp(short) 

250 

251 # lines[1] is UTC, lines[2] is EST, lines[3] is MST, lines[4] is PST 

252 return f"{lines[1]} ({lines[2]} / {lines[3]} / {lines[4]})"

255def get_multiline_timestamp(short: str) -> list[str]: 

256 """ 

257 Given a timestamp string from CI/CD in the form of 

258 20250815_211112_UTC, return a list of strings for multiple timezones. 

259 

260 Args: 

261 short: the UTC bash string, for example: 20250815_211112_UTC 

262 

263 Returns: 

264 List of 4 formatted strings. 

265 """ 

266 # Regex pattern to match the required format: YYYYMMDD_HHMMSS_TZ 

267 pattern: re.Pattern = re.compile(r"^(\d{8})_(\d{6})_(UTC|GMT|Z)$") 

268 match = pattern.match(short) 

269 

270 if not match: 

271 raise ValueError(f"Invalid timestamp format: '{short}'") 

272 

273 # Extract the date and time parts from the regex match 

274 date_part, time_part, _ = match.groups() 

275 

276 # Combine the parts into a format that can be parsed by datetime 

277 datetime_str: str = f"{date_part}_{time_part}_UTC" 

278 input_format: str = "%Y%m%d_%H%M%S_%Z" 

279 

280 # Convert the input string to a datetime object 

281 utc_datetime: datetime = datetime.strptime(datetime_str, input_format) 

282 

283 # Make the datetime object timezone-aware 

284 utc_now: datetime = pytz.utc.localize(utc_datetime) 

285 

286 # Get the formatted strings for each timezone 

287 utc_str, est_str, mst_str, pst_str = _get_timezone_strings(utc_now) 

288 

289 return ["Generated:", utc_str, est_str, mst_str, pst_str]

292def expiry_date(short: str) -> str: 

293 """ 

294 Return the GitHub Actions run expiry date as YYYY-mm-dd. 

295 

296 GitHub retains workflow run logs for GITHUB_RUN_EXPIRY_DAYS days. 

297 

298 Args: 

299 short: UTC timestamp in the form YYYYMMDD_HHMMSS_UTC 

300 

301 Returns: 

302 Expiry date string in YYYY-mm-dd format. 

303 """ 

304 pattern: re.Pattern = re.compile(r"^(\d{8})_(\d{6})_(UTC|GMT|Z)$") 

305 match = pattern.match(short) 

306 if not match: 

307 raise ValueError(f"Invalid timestamp format: '{short}'") 

308 date_part, time_part, _ = match.groups() 

309 utc_dt: datetime = datetime.strptime( 

310 f"{date_part}_{time_part}_UTC", "%Y%m%d_%H%M%S_%Z" 

311 ) 

312 expiry: datetime = utc_dt + timedelta(days=GITHUB_RUN_EXPIRY_DAYS) 

313 return expiry.strftime("%Y-%m-%d")
