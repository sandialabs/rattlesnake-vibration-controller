167 def add_environment(self): 

168 """Adds a row to the environment table""" 

169 selected_row = self.environment_display_table.rowCount() 

170 self.environment_display_table.insertRow(selected_row) 

171 combobox = QtWidgets.QComboBox() 

172 for control_type in combined_environments_capable: 

173 combobox.addItem(control_type.name.title(), control_type.value) 

174 self.environment_display_table.setCellWidget(selected_row, 0, combobox) 

175 

176 def remove_environment(self): 

177 """Removes a row from the environment table""" 

178 selected_row = self.environment_display_table.currentRow() 

179 if selected_row >= 0: 

180 self.environment_display_table.removeRow(selected_row)

182 def save_profile_template(self): 

183 """Saves a template for the given environments table 

184 

185 This template can be filled out by a user and then loaded.""" 

186 filename, _ = QtWidgets.QFileDialog.getSaveFileName( 

187 self, "Save Combined Environments Template", filter="Excel File (*.xlsx)" 

188 ) 

189 if filename == "": 

190 return 

191 # Now do the environments 

192 environment_data = [] 

193 for row in range(self.environment_display_table.rowCount()): 

194 combobox = self.environment_display_table.cellWidget(row, 0) 

195 value = ControlTypes(combobox.currentData()) 

196 name = self.environment_display_table.item(row, 1).text() 

197 environment_data.append((value, name)) 

198 save_combined_environments_profile_template(filename, environment_data)

212 def select_environment(parent=None): 

213 """Creates the dialog box and then parses the output. 

214 

215 Note that there are variable numbers of outputs for this function 

216 

217 Parameters 

218 ---------- 

219 parent : QWidget 

220 Parent to the dialog box (Default value = None) 

221 

222 Returns 

223 ------- 

224 result : int 

225 A flag specifying the outcome of the dialog box. Will be 1 if the 

226 dialog was accepted, zero if cancelled, and -1 if a profile was 

227 loaded instead. 

228 environment_table : list of lists 

229 A list of environment type, environment name pairs that will be 

230 used to specify the environments in a test. 

231 loaded_profile : str 

232 File name to the profile file that needs to be loaded. Only 

233 output if result == -1 

234 """ 

235 dialog = EnvironmentSelect(parent) 

236 result = 1 if (dialog.exec_() == QtWidgets.QDialog.Accepted) else 0 

237 if dialog.loaded_profile is None: 

238 environment_table = [] 

239 if result: 

240 for row in range(dialog.environment_display_table.rowCount()): 

241 combobox = dialog.environment_display_table.cellWidget(row, 0) 

242 value = ControlTypes(combobox.currentData()) 

243 name = dialog.environment_display_table.item(row, 1).text() 

244 environment_table.append([value, name]) 

245 # print(environment_table) 

246 return result, environment_table 

247 else: 

248 result = -1 

249 workbook = openpyxl.load_workbook(dialog.loaded_profile) 

250 environment_sheets = [ 

251 sheet 

252 for sheet in workbook 

253 if (sheet.title not in ["Channel Table", "Hardware", "Test Profile"]) 

254 and sheet.cell(1, 1).value == "Control Type" 

255 ] 

256 environment_table = [ 

257 (ControlTypes[sheet.cell(1, 2).value.upper()], sheet.title) 

258 for sheet in environment_sheets 

259 ] 

260 workbook.close() 

261 return result, environment_table, dialog.loaded_profile
