130def sine_sweep( 

131 dt, 

132 frequencies, 

133 sweep_rates, 

134 sweep_types, 

135 amplitudes=1, 

136 phases=0, 

137 return_argument=False, 

138 return_frequency=False, 

139 return_amplitude=False, 

140 return_phase=False, 

141 return_abscissa=False, 

142 only_breakpoints=False, 

143): 

144 """ 

145 Generates a sweeping sine wave with linear or logarithmic sweep rate 

146 

147 Parameters 

148 ---------- 

149 dt : float 

150 The time step of the output signal 

151 frequencies : iterable 

152 A list of frequency breakpoints for the sweep. Can be ascending or 

153 decending or both. Frequencies are specified in Hz, not rad/s. 

154 sweep_rates : iterable 

155 A list of sweep rates between the breakpoints. This array should have 

156 one fewer element than the `frequencies` array. The ith element of this 

157 array specifies the sweep rate between `frequencies[i]` and 

158 `frequencies[i+1]`. For a linear sweep, 

159 the rate is in Hz/s. For a logarithmic sweep, the rate is in octave/s. 

160 sweep_types : iterable or str 

161 The type of sweep to perform between each frequency breakpoint. Can be 

162 'lin' or 'log'. If a string is specified, it will be used for all 

163 breakpoints. Otherwise it should be an array containing strings with 

164 one fewer element than that of the `frequencies` array. 

165 amplitudes : iterable or float, optional 

166 Amplitude of the cosine wave at each of the frequency breakpoints. Can 

167 be specified as a single floating point value, or as an array with a 

168 value specified for each breakpoint. The default is 1. 

169 phases : iterable or float, optional 

170 Phases in radians of the cosine wave at each of the frequency breakpoints. Can 

171 be specified as a single floating point value, or as an array with a 

172 value specified for each breakpoint. Be aware that modifying the phase 

173 between breakpoints will effectively change the frequency of the signal, 

174 because the phase will change over time. The default is 0. 

175 return_argument : bool 

176 If True, return cosine argument over time 

177 return_frequency : bool 

178 If True, return the instantaneous frequency over time 

179 return_amplitude : bool 

180 If True, return the instantaneous amplitude over time 

181 return_phase : bool 

182 If True, return the instantaneous phase over time 

183 return_abscissa : bool 

184 If True, return the instantaneous abscissa over time 

185 only_breakpoints : bool 

186 If True, only returns data at breakpoints. Default is False 

187 

188 Raises 

189 ------ 

190 ValueError 

191 If the sweep rate and start and end frequency would result in a negative 

192 sweep time, for example if the start frequency is above the end frequency 

193 and a positive sweep rate is specified. 

194 

195 Returns 

196 ------- 

197 ordinate : np.ndarray 

198 A numpy array consisting of the generated sine sweep signal. The length 

199 of the signal will be determined by the frequency breakpoints and sweep 

200 rates. 

201 arg_over_time : np.ndarray 

202 A numpy array consisting of the argument to the cosine wave over time. 

203 freq_over_time : np.ndarray 

204 A numpy array consisting of the frequency of the cosine wave over time. 

205 amp_over_time : np.ndarray 

206 A numpy array consistsing of the amplitude of the cosine wave over time. 

207 phs_over_time : np.ndarray 

208 A numpy array consisting of the added phase in radians of the cosine 

209 wave over time. 

210 abscissa : np.ndarray 

211 A numpy array consisting of the time value at each time step returned 

212 

213 """ 

214 last_phase = 0 

215 last_abscissa = 0 

216 abscissa = [] 

217 ordinate = [] 

218 arg_over_time = [] 

219 freq_over_time = [] 

220 amp_over_time = [] 

221 phs_over_time = [] 

222 

223 # Go through each section 

224 for i in range(len(frequencies) - 1): 

225 # Extract the terms 

226 start_frequency = frequencies[i] 

227 end_frequency = frequencies[i + 1] 

228 omega_start = start_frequency * 2 * np.pi 

229 try: 

230 sweep_rate = sweep_rates[i] 

231 except TypeError: 

232 sweep_rate = sweep_rates 

233 if isinstance(sweep_types, str): 

234 sweep_type = sweep_types 

235 else: 

236 sweep_type = sweep_types[i] 

237 try: 

238 start_amplitude = amplitudes[i] 

239 end_amplitude = amplitudes[i + 1] 

240 except TypeError: 

241 start_amplitude = amplitudes 

242 end_amplitude = amplitudes 

243 try: 

244 start_phase = phases[i] # Radians 

245 end_phase = phases[i + 1] # Radians 

246 except TypeError: 

247 start_phase = phases # Radians 

248 end_phase = phases # Radians 

249 # Compute the length of this portion of the signal 

250 if sweep_type.lower() in ["lin", "linear"]: 

251 sweep_time = +(end_frequency - start_frequency) / sweep_rate 

252 elif sweep_type.lower() in ["log", "logarithmic"]: 

253 sweep_time = np.log(end_frequency / start_frequency) / ( 

254 sweep_rate * np.log(2) 

255 ) 

256 else: 

257 raise ValueError( 

258 "Sweep type should be one of lin, linear, log, or logarithmic" 

259 ) 

260 if sweep_time < 0: 

261 raise ValueError( 

262 f"Sweep time for segment index {i} is negative. Check sweep rate." 

263 ) 

264 sweep_samples = int(np.floor(sweep_time / dt)) 

265 # Construct the abscissa 

266 if only_breakpoints: 

267 this_abscissa = np.array([0, sweep_samples * dt]) 

268 else: 

269 this_abscissa = np.arange(sweep_samples + 1) * dt 

270 # Compute the phase over time 

271 if sweep_type.lower() in ["lin", "linear"]: 

272 this_argument = (1 / 2) * ( 

273 sweep_rate * 2 * np.pi 

274 ) * this_abscissa**2 + omega_start * this_abscissa 

275 this_frequency = (sweep_rate) * this_abscissa + omega_start / (2 * np.pi) 

276 elif sweep_type.lower() in ["log", "logarithmic"]: 

277 this_argument = 2 ** (sweep_rate * this_abscissa) * omega_start / ( 

278 sweep_rate * np.log(2) 

279 ) - omega_start / (sweep_rate * np.log(2)) 

280 this_frequency = ( 

281 2 ** (sweep_rate * this_abscissa) * omega_start / (2 * np.pi) 

282 ) 

283 else: 

284 raise ValueError( 

285 "Invalid sweep type, should be linear, lin, logarithmic, or log" 

286 ) 

287 # Compute the phase at each time step 

288 if end_frequency > start_frequency: 

289 freq_interp = [start_frequency, end_frequency] 

290 phase_interp = [start_phase, end_phase] 

291 amp_interp = [start_amplitude, end_amplitude] 

292 else: 

293 freq_interp = [end_frequency, start_frequency] 

294 phase_interp = [end_phase, start_phase] 

295 amp_interp = [end_amplitude, start_amplitude] 

296 this_phases = np.interp(this_frequency, freq_interp, phase_interp) 

297 # Compute the amplitude at each time step 

298 this_amplitudes = np.interp(this_frequency, freq_interp, amp_interp) 

299 this_ordinate = this_amplitudes * np.cos( 

300 this_argument + this_phases + last_phase 

301 ) 

302 if i == len(frequencies) - 2: 

303 last_index = None # If it's the last segment, go up until the end 

304 else: 

305 last_index = ( 

306 -1 

307 ) # Otherwise, we remove the last point because the first point of the 

308 # next segment will be this value 

309 arg_over_time.append(this_argument[:last_index] + last_phase) 

310 last_phase += this_argument[-1] 

311 abscissa.append(this_abscissa[:last_index] + last_abscissa) 

312 last_abscissa += this_abscissa[-1] 

313 ordinate.append(this_ordinate[:last_index]) 

314 freq_over_time.append(this_frequency[:last_index]) 

315 amp_over_time.append(this_amplitudes[:last_index]) 

316 phs_over_time.append(this_phases[:last_index]) 

317 ordinate = np.concatenate(ordinate) 

318 return_vals = [ordinate] 

319 if return_argument: 

320 return_vals.append(np.concatenate(arg_over_time)) 

321 if return_frequency: 

322 return_vals.append(np.concatenate(freq_over_time)) 

323 if return_amplitude: 

324 return_vals.append(np.concatenate(amp_over_time)) 

325 if return_phase: 

326 return_vals.append(np.concatenate(phs_over_time)) 

327 if return_abscissa: 

328 return_vals.append(np.concatenate(abscissa)) 

329 if len(return_vals) == 1: 

330 return_vals = return_vals[0] 

331 else: 

332 return_vals = tuple(return_vals) 

333 return return_vals

447def vold_kalman_filter( 

448 sample_rate, 

449 signal, 

450 arguments, 

451 filter_order=None, 

452 bandwidth=None, 

453 method=None, 

454 return_amp_phs=False, 

455 return_envelope=False, 

456 return_r=False, 

457): 

458 """ 

459 Extract sinusoidal components from a signal using the second generation 

460 Vold-Kalman filter. 

461 

462 Parameters 

463 ---------- 

464 sample_rate : float 

465 The sample rate of the signal in Hz. 

466 signal : ndarray 

467 A 1D signal containing sinusoidal components that need to be extracted 

468 arguments : ndarray 

469 A 2D array consisting of the arguments to the sinusoidal components of 

470 the form exp(1j*argument). This is the integral over time of the 

471 angular frequency, which can be approximated as 

472 2*np.pi*scipy.integrate.cumulative_trapezoid(frequencies,timesteps,initial=0) 

473 if frequencies is the frequency at each time step in Hz timesteps is 

474 the vector of time steps in seconds. This is a 2D array where the 

475 number of rows is the 

476 number of different sinusoidal components that are desired to be 

477 extracted, and the number of columns are the number of time steps in 

478 the `signal` argument. 

479 filter_order : int, optional 

480 The order of the VK filter, which should be 1, 2, or 3. The default is 

481 2. The low-pass filter roll-off is approximately -40 dB per times the 

482 filter order. 

483 bandwidth : ndarray, optional 

484 The prescribed bandwidth of the filter. This is related to the filter 

485 selectivity parameter `r` in the literature. This will be broadcast to 

486 the same shape as the `arguments` argument. The default is the sample 

487 rate divided by 1000. 

488 method : str, optional 

489 Can be set to either 'single' or 'multi'. In a 'single' solution, each 

490 sinusoidal component will be solved independently without any coupling. 

491 This can be more efficient, but will result in errors if the 

492 frequencies of the sine waves cross. The 'multi' solution will solve 

493 for all sinusoidal components simultaneously, resulting in a better 

494 estimate of crossing frequencies. The default is 'multi'. 

495 return_amp_phs : bool 

496 Returns the amplitude and phase of the reconstructed signals at each 

497 time step. Default is False 

498 return_envelope : bool 

499 Returns the complex envelope and phasors at each time step. Default is 

500 False 

501 return_r : bool 

502 Returns the computed selectivity parameters for the filter. Default is 

503 False 

504 

505 Raises 

506 ------ 

507 ValueError 

508 If arguments are not the correct size or values. 

509 

510 Returns 

511 ------- 

512 reconstructed_signals : ndarray 

513 Returns a time history the same size as `signal` for each of the 

514 sinusoidal components solved for. 

515 reconstructed_amplitudes : ndarray 

516 Returns the amplitude over time for each of the sinusoidal components 

517 solved for. Only returned if return_amp_phs is True. 

518 reconstructed_phases : ndarray 

519 Returns the phase over time for each of the sinusoidal components 

520 solved for. Only returned if return_amp_phs is True. 

521 reconstructed_envelope : ndarray 

522 Returns the complex envelope `x` over time for each of the sinusoidal 

523 components solved for. Only returned if return_envelope is True. 

524 reconstructed_phasor : ndarray 

525 Returns the phasor `c` over time for each of the sinusoidal components 

526 solved for. Only returned if return_envelope is True. 

527 r : ndarray 

528 Returns the selectivity `r` over time for each of the sinusoidal 

529 components solved for. Only returned if return_r is True. 

530 

531 """ 

532 # pylint: disable=invalid-name 

533 if filter_order is None: 

534 filter_order = 2 

535 

536 if bandwidth is None: 

537 bandwidth = sample_rate / 1000 

538 

539 # Make sure input data are numpy arrays 

540 signal = np.array(signal) 

541 arguments = np.atleast_2d(arguments) 

542 bandwidth = np.atleast_2d(bandwidth) 

543 bandwidth = np.broadcast_to(bandwidth, arguments.shape) 

544 relative_bandwidth = bandwidth / sample_rate 

545 

546 # Extract some sizes to make sure everything is correctly sized 

547 n_samples = signal.shape[-1] 

548 

549 n_orders_arg, n_arg = arguments.shape 

550 if n_arg != n_samples: 

551 raise ValueError( 

552 "Argument array must have identical number of columns as samples in signal" 

553 ) 

554 

555 if method is None: 

556 if n_orders_arg > 1: 

557 method = "multi" 

558 else: 

559 method = "single" 

560 if method.lower() not in ["multi", "single"]: 

561 raise ValueError('`method` must be either "multi" or "single"') 

562 

563 # Construct phasors to multiply the signals by 

564 phasor = np.exp(1j * arguments) 

565 

566 # Construct the matrices for the least squares solution 

567 if filter_order == 1: 

568 coefs = np.array([1, -1]) 

569 r = np.sqrt((np.sqrt(2) - 1) / (2 * (1 - np.cos(np.pi * relative_bandwidth)))) 

570 elif filter_order == 2: 

571 coefs = np.array([1, -2, 1]) 

572 r = np.sqrt( 

573 (np.sqrt(2) - 1) 

574 / ( 

575 6 

576 - 8 * np.cos(np.pi * relative_bandwidth) 

577 + 2 * np.cos(2 * np.pi * relative_bandwidth) 

578 ) 

579 ) 

580 elif filter_order == 3: 

581 coefs = np.array([1, -3, 3, -1]) 

582 r = np.sqrt( 

583 (np.sqrt(2) - 1) 

584 / ( 

585 20 

586 - 30 * np.cos(np.pi * relative_bandwidth) 

587 + 12 * np.cos(2 * np.pi * relative_bandwidth) 

588 - 2 * np.cos(3 * np.pi * relative_bandwidth) 

589 ) 

590 ) 

591 else: 

592 raise ValueError("filter order must be 1, 2, or 3") 

593 

594 # Construct the solution matrices 

595 A = sparse.spdiags( 

596 np.tile(coefs, (n_samples, 1)).T, 

597 np.arange(filter_order + 1), 

598 n_samples - filter_order, 

599 n_samples, 

600 ) 

601 B = [] 

602 for rvec in r: 

603 R = sparse.spdiags(rvec, 0, n_samples, n_samples) 

604 AR = A @ R 

605 B.append((AR).T @ (AR) + sparse.eye(n_samples)) 

606 

607 if method.lower() == "multi": 

608 # This solves the multiple order approach, constructing a big matrix of 

609 # Bs on the diagonal and CHCs on the off-diagonals. We can set up the 

610 # matrix as diagonals and upper diagonals then add the transpose to get the 

611 # lower diagonals 

612 B_multi_diagonal = sparse.block_diag(B) 

613 # There will be number of orders**2 B matrices, and number of orders 

614 # diagonals, so there will be n_orders**2-n_orders off diagonals, half on 

615 # on the upper triangle. We need to fill in all of these values for all 

616 # time steps. 

617 num_off_diags = (n_orders_arg**2 - n_orders_arg) // 2 

618 row_indices = np.zeros((n_samples, num_off_diags), dtype=int) 

619 col_indices = np.zeros((n_samples, num_off_diags), dtype=int) 

620 CHC = np.zeros((n_samples, num_off_diags), dtype="c16") 

621 # Keep track of the off-diagonal index so we know which column to put the 

622 # data in 

623 off_diagonal_index = 0 

624 # Now we need to step through the off-diagonal blocks and create the arrays 

625 for row_index in range(n_orders_arg): 

626 # Since we need to stay on the upper triangle, column indices will start 

627 # after the diagonal entry 

628 for col_index in range(row_index + 1, n_orders_arg): 

629 row_indices[:, off_diagonal_index] = np.arange( 

630 row_index * n_samples, (row_index + 1) * n_samples 

631 ) 

632 col_indices[:, off_diagonal_index] = np.arange( 

633 col_index * n_samples, (col_index + 1) * n_samples 

634 ) 

635 CHC[:, off_diagonal_index] = ( 

636 phasor[row_index].conj() * phasor[col_index] 

637 ) 

638 off_diagonal_index += 1 

639 # We set up the variables as multidimensional so we could store them easier, 

640 # but now we need to flatten them to put them into the sparse matrix. 

641 # We choose CSR because we can do math with it easier 

642 B_multi_utri = sparse.csr_matrix( 

643 (CHC.flatten(), (row_indices.flatten(), col_indices.flatten())), 

644 shape=B_multi_diagonal.shape, 

645 ) 

646 

647 # Now we can assemble the entire matrix by adding with the complex conjugate 

648 # of the upper triangle to get the lower triangle 

649 B_multi = B_multi_diagonal + B_multi_utri + B_multi_utri.getH() 

650 

651 # We also need to construct the right hand side of the equation. This 

652 # should be a multiplication of the phasor^H with the signal 

653 RHS = phasor.flatten().conj() * np.tile(signal, n_orders_arg) 

654 

655 x_multi = linalg.spsolve(B_multi, RHS[:, np.newaxis]) 

656 x = 2 * x_multi.reshape( 

657 n_orders_arg, -1 

658 ) # Multiply by 2 to account for missing negative frequency components 

659 else: 

660 # This solves the single order approach. If the user has put in multiple 

661 # orders, it will solve them all independently instead of combining them 

662 # into a single larger solve. 

663 x = np.zeros((n_orders_arg, n_samples), dtype=np.complex128) 

664 for i, (phasor_i, B_i) in enumerate(zip(phasor, B)): 

665 # We already have the left side of the equation B, now we just need the 

666 # right side of the equation, which is the phasor hermetian 

667 # times the signal elementwise-multiplied 

668 RHS = phasor_i.conj() * signal 

669 x[i] = 2 * linalg.spsolve(B_i, RHS) 

670 

671 return_value = [np.real(x * phasor)] 

672 if return_amp_phs: 

673 return_value.extend([np.abs(x), np.angle(x)]) 

674 if return_envelope: 

675 return_value.extend([x, phasor]) 

676 if return_r: 

677 return_value.extend([r]) 

678 if len(return_value) == 1: 

679 return return_value[0] 

680 else: 

681 return return_value

684def vold_kalman_filter_generator( 

685 sample_rate, 

686 num_orders, 

687 block_size, 

688 overlap, 

689 filter_order=None, 

690 bandwidth=None, 

691 method=None, 

692 buffer_size_factor=3, 

693): 

694 """ 

695 Extracts sinusoidal information using a Vold-Kalman Filter 

696 

697 This uses an windowed-overlap-and-add process to solve for the signal while 

698 removing start and end effects of the filter. Each time the generator is 

699 called, it will yield a further section of the results up until the overlap 

700 section. 

701 

702 Parameters 

703 ---------- 

704 sample_rate : float 

705 The sample rate of the signal in Hz. 

706 num_orders : int 

707 The number of orders that will be found in the signal 

708 block_size : int 

709 The size of the blocks used in the analysis. 

710 overlap : float, optional 

711 Fraction of the block size to overlap when computing the results. If 

712 not specified, it will default to 0.15. 

713 filter_order : int, optional 

714 The order of the VK filter, which should be 1, 2, or 3. The default is 

715 2. The low-pass filter roll-off is approximately -40 dB per times the 

716 filter order. 

717 bandwidth : ndarray, optional 

718 The prescribed bandwidth of the filter. This is related to the filter 

719 selectivity parameter `r` in the literature. This will be broadcast to 

720 the same shape as the `arguments` argument. The default is the sample 

721 rate divided by 1000. 

722 method : str, optional 

723 Can be set to either 'single' or 'multi'. In a 'single' solution, each 

724 sinusoidal component will be solved independently without any coupling. 

725 This can be more efficient, but will result in errors if the 

726 frequencies of the sine waves cross. The 'multi' solution will solve 

727 for all sinusoidal components simultaneously, resulting in a better 

728 estimate of crossing frequencies. The default is 'multi'. 

729 buffer_size_factor : int, optional 

730 Specifies the size of the buffer. buffer_size_factor * block_size is 

731 the size of the buffer. 

732 

733 Raises 

734 ------ 

735 ValueError 

736 If arguments are not the correct size or values. 

737 ValueError 

738 If data is provided subsequently to specifying last_signal = True 

739 

740 Sends 

741 ----- 

742 xi : iterable 

743 The next block of the signal to be filtered. This should be a 1D 

744 signal containing sinusoidal components that need to be extracted. 

745 argsi : iterable 

746 A 2D array consisting of the arguments to the sinusoidal components of 

747 the form exp(1j*argsi). This is the integral over time of the 

748 angular frequency, which can be approximated as 

749 2*np.pi*scipy.integrate.cumulative_trapezoid(frequencies,timesteps,initial=0) 

750 if frequencies is the frequency at each time step in Hz timesteps is 

751 the vector of time steps in seconds. This is a 2D array where the 

752 number of rows is the 

753 number of different sinusoidal components that are desired to be 

754 extracted, and the number of columns are the number of time steps in 

755 the `signal` argument. 

756 last_signal : bool 

757 If True, the remainder of the data will be returned and the 

758 overlap-and-add process will be finished. 

759 

760 Yields 

761 ------- 

762 reconstructed_signals : ndarray 

763 Returns a time history the same size as `signal` for each of the 

764 sinusoidal components solved for. 

765 reconstructed_amplitudes : ndarray 

766 Returns the amplitude over time for each of the sinusoidal components 

767 solved for. Only returned if return_amp_phs is True. 

768 reconstructed_phases : ndarray 

769 Returns the phase over time for each of the sinusoidal components 

770 solved for. Only returned if return_amp_phs is True. 

771 

772 """ 

773 previous_envelope = None 

774 reconstructed_signals = None 

775 reconstructed_amplitudes = None 

776 reconstructed_phases = None 

777 overlap_samples = int(overlap * block_size) 

778 window = windows.hann(overlap_samples * 2, False) 

779 start_window = window[:overlap_samples] 

780 end_window = window[overlap_samples:] 

781 buffer = CircularBufferWithOverlap( 

782 buffer_size_factor * block_size, 

783 block_size, 

784 overlap_samples, 

785 data_shape=(num_orders + 1,), 

786 ) 

787 first_output = True 

788 last_signal = False 

789 while True: 

790 xi, argsi, check_last_signal = ( 

791 yield reconstructed_signals, 

792 reconstructed_amplitudes, 

793 reconstructed_phases, 

794 ) 

795 if last_signal and check_last_signal: 

796 raise ValueError("Generator has been exhausted.") 

797 last_signal = check_last_signal 

798 argsi = np.atleast_2d(argsi) 

799 buffer_data = np.concatenate([xi[np.newaxis], argsi]) 

800 # print(f"{buffer_data.shape=}") 

801 buffer_output = buffer.write_get_data(buffer_data, last_signal) 

802 if buffer_output is not None: 

803 # print(f"{buffer_output.shape=}") 

804 if first_output: 

805 buffer_output = buffer_output[..., overlap_samples:] 

806 first_output = False 

807 signal = buffer_output[0] 

808 arguments = buffer_output[1:] 

809 else: 

810 signal = buffer_output[0] 

811 arguments = buffer_output[1:] 

812 signal[:overlap_samples] = signal[:overlap_samples] * start_window 

813 if not last_signal: 

814 signal[-overlap_samples:] = signal[-overlap_samples:] * end_window 

815 # print(f"{signal.shape=}") 

816 # Do the VK Filtering 

817 _, vk_envelope, vk_phasor = vold_kalman_filter( 

818 sample_rate, 

819 signal, 

820 arguments, 

821 filter_order, 

822 bandwidth, 

823 method, 

824 return_envelope=True, 

825 ) 

826 # print(f"{vk_signal.shape=}") 

827 # If necessary, do the overlap 

828 if previous_envelope is not None: 

829 vk_envelope[..., :overlap_samples] = ( 

830 vk_envelope[..., :overlap_samples] 

831 + previous_envelope[..., -overlap_samples:] 

832 ) 

833 if not last_signal: 

834 reconstructed_signals = np.real( 

835 vk_envelope[..., :-overlap_samples] 

836 * vk_phasor[..., :-overlap_samples] 

837 ) 

838 reconstructed_amplitudes = np.abs(vk_envelope[..., :-overlap_samples]) 

839 reconstructed_phases = np.angle(vk_envelope[..., :-overlap_samples]) 

840 else: 

841 reconstructed_signals = np.real(vk_envelope * vk_phasor) 

842 reconstructed_amplitudes = np.abs(vk_envelope) 

843 reconstructed_phases = np.angle(vk_envelope) 

844 previous_envelope = vk_envelope 

845 else: 

846 # print(f"{buffer_output=}") 

847 reconstructed_signals = None 

848 reconstructed_amplitudes = None 

849 reconstructed_phases = None

852class CircularBufferWithOverlap: 

853 """ 

854 A Circular buffer that allows data to be added and removed 

855 from the buffer with overlap 

856 """ 

857 

858 def __init__( 

859 self, buffer_size, block_size, overlap_size, dtype="float", data_shape=() 

860 ): 

861 """Initialize the circular buffer 

862 

863 Parameters 

864 ---------- 

865 buffer_size : int 

866 The total size of the circular buffer 

867 block_size : int 

868 The number of samples written or read in each block 

869 overlap_size : int 

870 The number of samples from the previous block to include in the read. 

871 dtype : dtype, optional 

872 The type of data in the buffer. The default is "float". 

873 data_shape : tuple, optional 

874 The shape of data in the buffer. The default is (). 

875 """ 

876 self.buffer_size = buffer_size 

877 self.block_size = block_size 

878 self.overlap_size = overlap_size 

879 self.buffer = np.zeros( 

880 tuple(data_shape) + (buffer_size,), dtype=dtype 

881 ) # Initialize buffer with zeros 

882 self.buffer_read = np.ones((buffer_size,), dtype=bool) 

883 self.write_index = 0 # Index where the next block will be written 

884 self.read_index = 0 # Index where the next block will be read from 

885 self.debug = False 

886 if self.debug: 

887 self.report_buffer_state() 

888 

889 def report_buffer_state(self): 

890 """Prints the current buffer state""" 

891 read_samples = np.sum(self.buffer_read) 

892 write_samples = self.buffer_size - read_samples 

893 print(f"{read_samples} of {self.buffer_size} have been read") 

894 print(f"{write_samples} of {self.buffer_size} have been written but not read") 

895 

896 def write_get_data(self, data, read_remaining=False): 

897 """ 

898 Writes a block of data and then returns a block if available 

899 

900 Parameters: 

901 - data: Array to write to the buffer. 

902 """ 

903 self.write(data) 

904 try: 

905 return self.read(read_remaining) 

906 except ValueError: 

907 return None 

908 

909 def write(self, data): 

910 """ 

911 Write a block of data to the circular buffer. 

912 

913 Parameters: 

914 - data: Array to write to the buffer. 

915 """ 

916 # Compute the end index for the write operation 

917 indices = ( 

918 np.arange( 

919 self.write_index, self.write_index + data.shape[-1] + self.overlap_size 

920 ) 

921 % self.buffer_size 

922 ) 

923 

924 if np.any(~self.buffer_read[indices]): 

925 raise ValueError( 

926 "Overwriting data on buffer that has not been read. " 

927 "Read data before writing again." 

928 ) 

929 

930 self.buffer[ 

931 ..., indices[: None if self.overlap_size == 0 else -self.overlap_size] 

932 ] = data 

933 self.buffer_read[ 

934 indices[: None if self.overlap_size == 0 else -self.overlap_size] 

935 ] = False 

936 

937 # Update the write index 

938 self.write_index = (self.write_index + data.shape[-1]) % self.buffer_size 

939 

940 if self.debug: 

941 print("Wrote Data to Buffer") 

942 self.report_buffer_state() 

943 # print(self.buffer) 

944 # print(self.buffer_read) 

945 

946 def read(self, read_remaining=False): 

947 """ 

948 Reads data from the buffer 

949 

950 Parameters 

951 ---------- 

952 read_remaining : bool, optional 

953 If true, read everything left on the buffer that hasn't yet been 

954 read. The default is False. 

955 

956 Raises 

957 ------ 

958 ValueError 

959 If there is not a block of data on the buffer and data would be 

960 read a second time. 

961 

962 Returns 

963 ------- 

964 return_data : ndarray 

965 A block of data read from the buffer. 

966 

967 """ 

968 indices = ( 

969 np.arange( 

970 self.read_index - self.overlap_size, self.read_index + self.block_size 

971 ) 

972 % self.buffer_size 

973 ) 

974 if read_remaining: 

975 # Pick out just the indices that are ok to read 

976 # print('Reading Remaining:') 

977 # print(f"{indices.copy()=}") 

978 indices = np.concatenate( 

979 ( 

980 indices[: self.overlap_size], 

981 indices[self.overlap_size :][ 

982 ~self.buffer_read[indices[self.overlap_size :]] 

983 ], 

984 ) 

985 ) 

986 # print(f"{indices.copy()=}") 

987 if np.any(self.buffer_read[indices[self.overlap_size :]]): 

988 raise ValueError( 

989 "Data would be read multiple times. Write data before reading again." 

990 ) 

991 return_data = self.buffer[..., indices] 

992 self.buffer_read[indices[self.overlap_size :]] = True 

993 self.read_index = ( 

994 self.read_index + (return_data.shape[-1] - self.overlap_size) 

995 ) % self.buffer_size 

996 if self.debug: 

997 print("Read Data from Buffer") 

998 self.report_buffer_state() 

999 # print(self.buffer) 

1000 # print(self.buffer_read) 

1001 return return_data

1247 (np.zeros(delay_samples), np.linspace(0, 1, ramp_samples)) 

1248 ) 

1249 * amplitude[..., [0]] 

1250 ) 

1251 begin_phases = np.ones(ramp_samples + delay_samples) * phase[..., [0]] 

1252 begin_signal = begin_amplitudes * np.cos(begin_arguments + begin_phases) 

1253 

1254 abscissa = np.concatenate((begin_abscissa, abscissa), axis=-1) 

1255 ordinate = np.concatenate((begin_signal, ordinate), axis=-1) 

1256 frequency = np.concatenate((begin_frequencies, frequency), axis=-1) 

1257 amplitude = np.concatenate((begin_amplitudes, amplitude), axis=-1) 

1258 phase = np.concatenate((begin_phases, phase), axis=-1) 

1259 argument = np.concatenate((begin_arguments, argument), axis=-1) 

1260 if ramp_samples > 0: 

1261 end_abscissa = np.arange(1, ramp_samples + 1) / sample_rate 

1262 end_arguments = 2 * np.pi * frequency[-1] * end_abscissa + argument[-1] 

1263 end_frequencies = np.ones(ramp_samples) * frequency[-1] 

1264 end_amplitudes = np.linspace(1, 0, ramp_samples) * amplitude[..., [-1]] 

1265 end_phases = np.ones(ramp_samples) * phase[..., [-1]] 

1266 end_signal = end_amplitudes * np.cos(end_arguments + end_phases) 

1267 

1268 abscissa = np.concatenate((abscissa, end_abscissa), axis=-1) 

1269 ordinate = np.concatenate((ordinate, end_signal), axis=-1) 

1270 frequency = np.concatenate((frequency, end_frequencies), axis=-1) 

1271 argument = np.concatenate((argument, end_arguments), axis=-1) 

1272 amplitude = np.concatenate((amplitude, end_amplitudes), axis=-1) 

1273 phase = np.concatenate((phase, end_phases), axis=-1) 

1274 end_index = abscissa.shape[-1] - ramp_samples 

1275 

1276 return ( 

1277 ordinate, 

1278 frequency, 

1279 argument, 

1280 amplitude, 

1281 phase, 

1282 abscissa, 

1283 start_index, 

1284 end_index,

1412 ): 

1413 """ 

1414 Creates a combined signal from many specifications 

1415 

1416 Parameters 

1417 ---------- 

1418 specifications : list of SineSpecification objects 

1419 The various specification objects to combine together to creat the 

1420 combined signals. 

1421 sample_rate : float 

1422 The sample rate at which the signals should be generated. 

1423 ramp_samples : int 

1424 The number of samples to use in the ramp portion of the signals. 

1425 control_index : int, optional 

1426 The control channel index at which the signals should be generated. 

1427 The default is to generate all channels. 

1428 

1429 Returns 

1430 ------- 

1431 signal : ndarray 

1432 A number of channels by number of timesteps array of time history 

1433 values. 

1434 order_signals : ndarray 

1435 Separate signals for each of the specification tones, in a 

1436 num_tones, num_channels, num_timesteps array. 

1437 order_frequencies : ndarray 

1438 The frequencies associated with each sine tone at each time step 

1439 in a num_tones, num_timesteps array. 

1440 order_arguments : ndarray 

1441 The arguments associated with each sine tone at each time step in a 

1442 num_tones, num_timestep array 

1443 order_amplitudes : ndarray 

1444 The instantaneous amplitudes associated with each sine tone and 

1445 channel at each time step in a num_tones, num_channels, 

1446 num_timesteps shaped array 

1447 order_phases : ndarray 

1448 The instantaneous phase associated with each sine tone and 

1449 channel at each time step in a num_tones, num_channels, 

1450 num_timesteps shaped array. Phases in Radians. 

1451 order_start_samples : ndarray 

1452 The starting sample for each tone taking into account the start time 

1453 and the ramp samples 

1454 order_end_samples : ndarray 

1455 The end sample for each tone taking into account the ramp samples. 

1456 

1457 """ 

1458 order_signals = [] 

1459 order_arguments = [] 

1460 order_frequencies = [] 

1461 order_amplitudes = [] 

1462 order_phases = [] 

1463 order_start_samples = [] 

1464 order_end_samples = [] 

1465 longest_signal = 0 

1466 for spec in specifications: 

1467 ( 

1468 ordinate, 

1469 frequency, 

1470 argument, 

1471 amplitude, 

1472 phase, 

1473 _, 

1474 start_index, 

1475 end_index, 

1476 ) = spec.create_signal(sample_rate, ramp_samples, control_index) 

1477 order_signals.append(ordinate) 

1478 order_frequencies.append(frequency) 

1479 order_amplitudes.append(amplitude) 

1480 order_phases.append(phase) 

1481 order_arguments.append(argument) 

1482 order_start_samples.append(start_index) 

1483 order_end_samples.append(end_index) 

1484 

1485 if order_signals[-1].shape[-1] > longest_signal: 

1486 longest_signal = order_signals[-1].shape[-1] 

1487 

1488 # Now that we know the longest signals, we know how much we need to pad 

1489 # to make all signals the same length 

1490 for i, signal in enumerate(order_signals): 

1491 extra_samples = longest_signal - signal.shape[-1] 

1492 end_abscissa = np.arange(1, extra_samples + 1) / sample_rate

1495 + order_arguments[i][-1] 

1496 )

1500 end_signal = np.zeros(extra_samples) * signal[..., [-1]]

1502 order_frequencies[i] = np.concatenate( 

1503 (order_frequencies[i], end_frequencies), axis=-1 

1504 )

1506 (order_arguments[i], end_arguments), axis=-1

1509 (order_amplitudes[i], end_amplitudes), axis=-1 

1510 )

1514 order_frequencies = np.array(order_frequencies) 

1515 order_arguments = np.array(order_arguments) 

1516 order_amplitudes = np.array(order_amplitudes) 

1517 order_phases = np.array(order_phases) 

1518 order_start_samples = np.array(order_start_samples) 

1519 order_end_samples = np.array(order_end_samples) 

1520 signal = np.sum(order_signals, axis=0) 

1521
