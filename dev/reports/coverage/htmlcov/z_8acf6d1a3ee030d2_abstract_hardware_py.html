32# region: Acqusition 

33class HardwareAcquisition(ABC): 

34 """Abstract class defining the interface between the controller and acquisition 

35 

36 This class defines the interfaces between the controller and the 

37 data acquisition portion of the hardware. It is run by the Acquisition 

38 process, and must define how to get data from the test hardware into the 

39 controller.""" 

40 

41 @abstractmethod 

42 def set_up_data_acquisition_parameters_and_channels( 

43 self, test_data: DataAcquisitionParameters, channel_data: List[Channel] 

44 ): 

45 """ 

46 Initialize the hardware and set up channels and sampling properties 

47 

48 The function must create channels on the hardware corresponding to 

49 the channels in the test. It must also set the sampling rates. 

50 

51 Parameters 

52 ---------- 

53 test_data : DataAcquisitionParameters : 

54 A container containing the data acquisition parameters for the 

55 controller set by the user. 

56 channel_data : List[Channel] : 

57 A list of ``Channel`` objects defining the channels in the test 

58 

59 Returns 

60 ------- 

61 None. 

62 

63 """ 

64 

65 @abstractmethod 

66 def start(self): 

67 """Method to start acquiring data from the hardware""" 

68 

69 @abstractmethod 

70 def read(self) -> np.ndarray: 

71 """Method to read a frame of data from the hardware that returns 

72 an appropriately sized np.ndarray""" 

73 

74 @abstractmethod 

75 def read_remaining(self) -> np.ndarray: 

76 """Method to read the rest of the data on the acquisition from the hardware 

77 that returns an appropriately sized np.ndarray""" 

78 

79 @abstractmethod 

80 def stop(self): 

81 """Method to stop the acquisition""" 

82 

83 @abstractmethod 

84 def close(self): 

85 """Method to close down the hardware""" 

86 

87 @abstractmethod 

88 def get_acquisition_delay(self) -> int: 

89 """Get the number of samples between output and acquisition 

90 

91 This function is designed to handle buffering done in the output 

92 hardware, ensuring that all data written to the output is read by the 

93 acquisition. If a output hardware has a buffer, there may be a non- 

94 negligable delay between when output is written to the device and 

95 actually played out from the device."""

98# region: Output 

99class HardwareOutput(ABC): 

100 """Abstract class defining the interface between the controller and output 

101 

102 This class defines the interfaces between the controller and the 

103 output or source portion of the hardware. It is run by the Output 

104 process, and must define how to get write data to the hardware from the 

105 control system""" 

106 

107 @abstractmethod 

108 def set_up_data_output_parameters_and_channels( 

109 self, test_data: DataAcquisitionParameters, channel_data: List[Channel] 

110 ): 

111 """ 

112 Initialize the hardware and set up sources and sampling properties 

113 

114 The function must create channels on the hardware corresponding to 

115 the sources in the test. It must also set the sampling rates. 

116 

117 Parameters 

118 ---------- 

119 test_data : DataAcquisitionParameters : 

120 A container containing the data acquisition parameters for the 

121 controller set by the user. 

122 channel_data : List[Channel] : 

123 A list of ``Channel`` objects defining the channels in the test 

124 

125 Returns 

126 ------- 

127 None. 

128 

129 """ 

130 

131 @abstractmethod 

132 def start(self): 

133 """Method to start outputting data to the hardware""" 

134 

135 @abstractmethod 

136 def write(self, data): 

137 """Method to write a np.ndarray with a frame of data to the hardware""" 

138 

139 @abstractmethod 

140 def stop(self): 

141 """Method to stop the output""" 

142 

143 @abstractmethod 

144 def close(self): 

145 """Method to close down the hardware""" 

146 

147 @abstractmethod 

148 def ready_for_new_output(self) -> bool: 

149 """Method that returns true if the hardware should accept a new signal"""
